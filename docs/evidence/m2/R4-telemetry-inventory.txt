### (f) a table on the NVA that no one in this project wrote
### cmd: az vm run-command invoke -g rg-cne-lab -n vm-cne-hub --command-id RunShellScript --scripts 'nft list table ip security'

table ip security {
	chain OUTPUT {
		type filter hook output priority 150; policy accept;
		ip daddr 168.63.129.16 tcp dport 53 counter packets 0 bytes 0 accept
		ip daddr 168.63.129.16 ip protocol tcp skuid 0 counter packets 13951 bytes 4925297 accept
		ip daddr 168.63.129.16 ip protocol tcp ct state invalid,new counter packets 0 bytes 0 drop
	}
}
# Warning: table ip security is managed by iptables-nft, do not touch!

### note: rules for Azure's platform address 168.63.129.16 (wireserver): DNS allowed, TCP allowed for root only (13,951 packets so far), any other new TCP connection to it dropped. This matches what the Azure Linux VM agent sets up so that unprivileged processes cannot reach the platform endpoint; that the agent created it is inferred from the pattern, not traced. It sits on the output hook and does not touch forwarded traffic. It means the NVA's effective ruleset is not only what cloud-init-nva.yaml writes.
